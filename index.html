from flask import Flask, request import os import traceback app = Flask(__name__) # File where ONLY the 10-digit numbers will be stored LOG_FILE = "PhoneNumbers.txt" def load_existing_numbers(): """Read existing numbers into a set (one 10-digit number per line).""" if not os.path.exists(LOG_FILE): return set() numbers = set() with open(LOG_FILE, "r", encoding="utf-8") as f: for line in f: num = line.strip() if num: numbers.add(num) return numbers def save_number(number: str): """Append the 10-digit number to the file with NO timestamp.""" with open(LOG_FILE, "a", encoding="utf-8") as f: f.write(number + "\n") @app.route("/submit", methods=["POST"]) def submit(): try: number = request.form.get("number", "").strip() # ✅ Server-side validation: ONLY 10 digits if not number.isdigit() or len(number) != 10: return "Invalid number. Must be exactly 10 digits.", 400 existing = load_existing_numbers() # ✅ Block duplicates if number in existing: print(f"Duplicate ignored: {number}") return "Number already submitted." # ✅ Save ONLY the digits save_number(number) print(f"Saved: {number}") return "Number submitted successfully." except Exception: traceback.print_exc() return "Server error.", 500 if __name__ == "__main__": print("Starting server on port 5000...") app.run(host="0.0.0.0", port=5000)
